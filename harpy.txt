setcpm(130/4)

$: chord("<Am7 <Em7 Em9>>").voicing()
  .arp("<[0 1 2 3] <[4 3 2 1] [0 1 0 2]>>")
  .fast(2)
  .delay(0.4)
  .room(1)
  .orbit(2)

$: note("a4*4").transpose("0 12 <0 2> <7 -2>")
  // .ply(2)
  .struct("x(7,16)")
  .sound("sine")
  .decay(0.1)
  .room(1)
  .delay(0.4)
  .orbit(2)
  .gain(0.8)

$: n(run(8).slow(2)).scale("A1:minor")
  .orbit(2)

$: n("2(3, 8)")
  .off(1/8, x=> x.add(n("<-2 [-2@3 1]>")))
  .scale("A3:minor")
  .sound("gm_acoustic_guitar_nylon:2")
  .decay(0.3)
  .delay(0.5)
  .room(1)
  .gain(0.6)
  .orbit(2)

$: n(irand(8).segment(16))
  .scale("A6:minor:pentatonic")
  .sound("saw")
  .decay(0.05)
  .room(1)
  .gain(rand.range(0, 0.4))
  .delay(0.8)
  .orbit(2)

$: s("bd:4*4")
  .duck(2)
  .duckonset(0.01)
  .duckattack(0.1)
  .duckdepth(0.8)

$: sound("white*16").decay(0.01).gain(saw.range(0.2, 1).fast(4))
