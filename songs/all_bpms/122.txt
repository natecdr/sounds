const _scale = "A:major"

    x=>x.arp(irand(4).segment(16).rib(1, 1))
      .transpose(24).gain(0.3).glide(0.01)
  )

  .transpose(12)
  .room(0.5)

  .orbit(2)

$: "<5@2 0 1>".note().scale(_scale).transpose(-24).distort("3:0.3")
  .orbit(2)

  .orbit(2)
