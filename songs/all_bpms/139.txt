var bpm = 139
setcpm(bpm/4)

samples('github:natecdr/sounds')

const _scale = "D#:major"

pad: note(irand(5).segment(64).rib(1, 4)).scale("D#:major:pentatonic")
  .sometimesBy(0.02, x=>x.transpose(12))
  .sound("sine")
  .decay(1)
  .release(1)
  .attack(0.1)
  .room(1)
  .superimpose(x=>sound("crackle"))
  .orbit(2)

bass: note("<-2 0 1 2>".slow(2)).scale(_scale)
  .sound("sine")
  .octave(-1)
  .distort("1:0.6")
  .orbit(2)

gling: note(irand(7).segment(1).ply(6).add(run(6)).palindrome()).scale(_scale).slow(2)
  .transpose(12)
  .sound("gm_pizzicato_strings:1")
  .decay(0.2)
  .delay(0.5).delaysync(1/6)
  .room(1)
  .lpf(400)
  .gain(0.6)
  .orbit(2)

chords: "<0 2 <3 [3 2:M]> -2>".chordDegree(_scale).voicing().slow(2)
  .octave(-1)
  .sound("saw")
  .lpf(600)
  .tremolo(3).tremolosync("<8 6>".slow(2)).tremoloshape("cosine")
  .decay(3)
  .room(0.5)
  .gain(0.5)
  .orbit(2)

kick: sound("bd:0")
  .struct("<x [- x - x]>")
  .lpf(1000)
  .duck(2)

snare: sound("sd:1")
  .struct("- x")
  .room(0.5)
  .lpf(1000)

perc: sound("hh:2")
  .struct("[- x]*2")
  .room(0.5)
  .lpf(1000)
