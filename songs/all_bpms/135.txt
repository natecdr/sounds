const bpm = 135
setcpm(bpm / 4)

const _scale = "A:phrygian"

metro: sound("white!4").decay(0.05)

lead: note(stepcat([3, "[0 7 -]!4"], [1, "[0 7]"])).scale(_scale)
  .transpose("<0!3 3>")
  //.sometimesBy(0.4, x=>x.transpose(12))
  .transpose("-12")
  .decay(0.3)
  .cut(1)
  .sound("supersaw")
  .phaser(slider(1, 0, 1))
  .fm(4).fmh(0.5)
  .noise(0.1)
  .distRect(8)
  .clip(slider(1, 0, 1))
  .lpf(300)
  .lpenv(-4).lpa(0.1).lpd(0)
  .hpf(150)
  .room(0.3)
  //.delay(0.5)
  .fscope({color:"red"})
  .orbit(2)

bass: note("0(5, 16)").scale(_scale)
  .octave(-2)
  .transpose("<0!3 3>")
  .sound("saw")
  .distRect(4)
  .orbit(2)

ohat: sound("[hh oh]*4").cut(1).legato(0.5)

sd: sound("[- sd]*2").n(1)

kick: sound("bd:5*4")
  .duck(2)

_riser: sound("pulse*16").fm(time).fmh(time).orbit(2)
