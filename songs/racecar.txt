var bpm = 174
var cpm = bpm/4

samples('github:natecdr/Sounds/main')

setcpm(cpm)

$: n("[2 - [- 2] [- 2?]]")
  .sound("kicks")
  .duckorbit(2)
  .lastOf(4, x=>x.mask("1 1?"))

$: n("[- 0 - 0]")
  .sound("snares")
  .off(
    1/8, x=> x.gain(0.3).speed(1.5)
  )
  .lastOf(4, x=>x.hush())

$: n("[0|2]*8")
  .sound("hh")
  .sometimesBy(0.2, x=> x.speed(2))
  .lastOf(4, x=>x.hush())

$: note("<e1 [g1 f#1] c1 [g1 a1]>/2").ply(8)
  .sound("saw")
  .distRect(slider(3.64, 0, 10))
  .lpf(800)
  .lastOf(4, x=>x.distRect(6).disttype("diode").decay(0.2))
  .orbit(2)

$: note("[e3 g3 b3 e4]*4").palindrome()
  .transpose(12)
  .room(0.5)
  .sound("tri")
  .decay(0.1)
  .lastOf(4, x=> x.decay(0.5).room(1).delay(0.5).slow("4 | 2"))
  .widen(0.5)
  .orbit(2)
